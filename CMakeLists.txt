cmake_minimum_required(VERSION 3.16)
project(resonator LANGUAGES CXX)

add_library(resonator_core SHARED backend.cpp)
add_executable(resonator main.cpp)
foreach(target resonator_core resonator)
    target_compile_features(${target} PRIVATE cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
endforeach()
set_target_properties(resonator_core PROPERTIES PREFIX "")

include(CTest)
if(BUILD_TESTING)
    add_executable(resonator_tests tests.cpp)
    target_compile_features(resonator_tests PRIVATE cxx_std_17)
    if(MINGW)
        target_link_options(resonator_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME flow_invariants COMMAND resonator_tests)
endif()
